function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Fetch Main Dashboard Data var projectsSheet = ss.getSheetByName("DashboardData"); var projectsData = []; if (projectsSheet) { var pData = projectsSheet.getDataRange().getValues(); var pHeaders = pData[0]; for (var i = 1; i < pData.length; i++) { var row = {}; for (var j = 0; j < pHeaders.length; j++) { row[pHeaders[j]] = pData[i][j]; } projectsData.push(row); } } // 2. Fetch Priorities Data var prioritiesSheet = ss.getSheetByName("Priorities"); var prioritiesData = []; if (prioritiesSheet) { var prData = prioritiesSheet.getDataRange().getValues(); var prHeaders = prData[0]; for (var i = 1; i < prData.length; i++) { var row = {}; for (var j = 0; j < prHeaders.length; j++) { row[prHeaders[j]] = prData[i][j]; } prioritiesData.push(row); } } // Package both datasets into one JSON response var result = { projects: projectsData, priorities: prioritiesData }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // 3. The Vote Catcher (POST Request) function doPost(e) { try { var payload = JSON.parse(e.postData.contents); // Check if the signal is a vote if (payload.action === "vote" && payload.itemId) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Priorities"); var data = sheet.getDataRange().getValues(); var headers = data[0]; var voteColIndex = headers.indexOf("VoteCount"); if (voteColIndex === -1) { return ContentService.createTextOutput(JSON.stringify({ error: "VoteCount column not found" })).setMimeType(ContentService.MimeType.JSON); } // Find the specific item ID and add +1 to its VoteCount for (var i = 1; i < data.length; i++) { if (data[i][0] == payload.itemId) { // Column 0 is ItemID var currentVotes = data[i][voteColIndex]; var newVotes = (Number(currentVotes) || 0) + 1; sheet.getRange(i + 1, voteColIndex + 1).setValue(newVotes); return ContentService.createTextOutput(JSON.stringify({ success: true, newVotes: newVotes })) .setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ error: "Item not found" })).setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({ error: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
